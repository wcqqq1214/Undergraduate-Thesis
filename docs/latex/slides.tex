\section{可解释机器学习特征分析}
% ============================================================

\begin{frame}{研究方法：LightGBM + SHAP}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item 直方图算法：连续特征离散化，降低计算量
        \item 叶子优先生长策略（Leaf-wise）
        \item 基于梯度的单边抽样（GOSS）
        \item 互斥特征捆绑（EFB）
      \end{itemize}
      \vspace{0.5em}
      迭代更新公式：
      \begin{equation*}
        F_t(x) = F_{t-1}(x) + \eta f_t(x)
      \end{equation*}
    \end{column}
    \begin{column}{0.50\textwidth}
      {\small
      \textbf{SHAP 可解释性}

      特征 $j$ 的 Shapley 值：
      \begin{equation*}
        \textstyle
        \varphi_j = \sum_{S \subseteq N\setminus\{j\}}
          \frac{|S|!(N\!-\!|S|\!-\!1)!}{N!}
          \bigl[f(S\!\cup\!\{j\})-f(S)\bigr]
      \end{equation*}

      \vspace{0.1em}
      交互值（$j\neq k$）：
      \begin{equation*}
        \varphi_{j,k} = \!\sum_{S \subseteq N\setminus\{j,k\}}
          \frac{|S|!(N\!-\!|S|\!-\!2)!}{2(N\!-\!1)!}
          \nabla_{jk}(S)
      \end{equation*}
      }
    \end{column}
  \end{columns}
\end{frame}

\section{基于 LSTM 的位移预测模型}
% ============================================================

\begin{frame}{LSTM 网络原理}
  LSTM 通过门控机制捕捉长期依赖关系，核心计算如下：

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \textbf{遗忘门}
      \begin{equation*}
        \mathbf{f}_t = \sigma(\mathbf{W}_f [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_f)
      \end{equation*}
      \textbf{输入门}
      \begin{equation*}
        \mathbf{i}_t = \sigma(\mathbf{W}_i [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_i)
      \end{equation*}
      \begin{equation*}
        \tilde{\mathbf{C}}_t = \tanh(\mathbf{W}_C [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_C)
      \end{equation*}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \textbf{记忆单元更新}
      \begin{equation*}
        \mathbf{C}_t = \mathbf{f}_t \odot \mathbf{C}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{C}}_t
      \end{equation*}
      \textbf{输出门}
      \begin{equation*}
        \mathbf{o}_t = \sigma(\mathbf{W}_o [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_o)
      \end{equation*}
      \begin{equation*}
        \mathbf{h}_t = \mathbf{o}_t \odot \tanh(\mathbf{C}_t)
      \end{equation*}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{block}{输入设计}
    MJ9、MJ1、MJ3、ATU4 四点历史累积位移序列（时间窗口 2 天，特征维度 4）$\;\to\;$ 预测 MJ1 趋势项位移
  \end{block}
\end{frame}

\begin{frame}{基于多次运行的概率预测框架}
  \textbf{方法}：对 LSTM 进行 \alert{50 次独立训练}（不同随机种子），收集预测分布

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{统计量计算}

      均值预测：
      \begin{equation*}
        \bar{y}_t = \frac{1}{50}\sum_{i=1}^{50}\hat{y}_t^{(i)}
      \end{equation*}

      分位数预测区间：
      \begin{equation*}
        \hat{y}_t^{(p)} = \text{Percentile}\!\left(\{\hat{y}_t^{(i)}\},\, p\right)
      \end{equation*}
      构建 50\% 区间（25\%–75\%）和 90\% 区间（5\%–95\%）
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{区间评价指标}

      覆盖概率（PICP）：
      \begin{equation*}
        \text{PICP} = \frac{1}{n}\sum_{i=1}^{n} \mathbb{I}(L_i \leq y_i \leq U_i)
      \end{equation*}

      归一化区间宽度（NMPIW）：
      \begin{equation*}
        \text{NMPIW} = \frac{1}{n \cdot R}\sum_{i=1}^{n}(U_i - L_i)
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{网络结构与超参数}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{模型结构}
      \begin{itemize}
        \item LSTM 层 1：25 个隐藏单元，return\_sequences=True
        \item Dropout：0.3
        \item LSTM 层 2：15 个隐藏单元，L2 正则化（0.002）
        \item Dropout：0.3
        \item 全连接层：15 → 1（输出位移预测值）
        \item 总参数量：5,716
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{训练设置}
      \begin{itemize}
        \item 优化器：Adam，学习率 0.0005
        \item 损失函数：MSE
        \item 批大小：64，训练轮数：40
        \item 数据划分：训练集 80\% / 测试集 20\%
        \item 归一化：MinMaxScaler $[0,1]$
      \end{itemize}
      \vspace{0.5em}
      \begin{block}{硬件}
        NVIDIA GeForce RTX 3060 Laptop GPU\\
        单次训练约 1.5 分钟
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{基于概率预测的滑坡预警方法}
% ============================================================

\begin{frame}{传统预警方法的局限性}
  \begin{enumerate}
    \item \textbf{忽略预测不确定性}：基于点预测判断，预测值接近阈值时易误报或漏报
    \item \textbf{响应时效性不足}：速率/切线角指标属"滞后性指标"，预警窗口极短
    \item \textbf{单一阈值缺乏适应性}：固定阈值难以适应不同变形阶段的动态变化
  \end{enumerate}
  \vspace{0.8em}
  \begin{block}{本章解决方案}
    基于 LSTM 输出的\textbf{概率分布}，计算预测位移超过安全阈值的\alert{越限概率}，建立五级动态预警体系
  \end{block}
\end{frame}

\begin{frame}[shrink=15]{越限概率计算方法}
  \textbf{经验累积分布函数（ECDF）}：
  \begin{equation*}
    F_t(d) = \frac{1}{50} \sum_{i=1}^{50} \mathbb{I}\!\left(\hat{D}_t^{(i)} \leq d\right)
  \end{equation*}

  \textbf{越限概率}（预测位移超过安全阈值 $D_\text{thr}$ 的概率）：
  \begin{equation*}
    P_{\text{exceed},t}
    = 1 - F_t(D_{\text{thr}})
    = \frac{1}{50} \sum_{i=1}^{50} \mathbb{I}\!\left(\hat{D}_t^{(i)} > D_{\text{thr}}\right)
  \end{equation*}

  \begin{block}{基准阈值设定依据}
    \small
    \begin{itemize}
      \item 参考《地质灾害气象风险预警判据》及三峡库区规范
      \item 经 LightGBM 分类任务数据驱动验证：$D_\text{thr} = 0.3\,\text{mm}$
      \item 接近 MJ9 训练集 95\% 分位数，处于 MJ1/MJ3 的 75\%–90\% 分位数之间
    \end{itemize}
  \end{block}
\end{frame}

    \item \alert{库水位上升}（266.0→268.3\,m）：贡献约 \textbf{45\%}
    \item \alert{累计降雨}（65.6\,mm，含单日 54.6\,mm）：贡献约 \textbf{30\%}
    \item 历史位移累积效应：贡献约 \textbf{20\%}
    \item 其他因素（地下水、温度）：约 5\%
  \end{itemize}
  \vspace{0.3em}
  \begin{block}{结论}
    "双重不利工况"（库水位变化 + 强降雨）是触发高等级预警的主要机制
  \end{block}
\end{frame}

\begin{frame}{主要结论}
  \small
  \begin{enumerate}
    \item \textbf{特征分析}：LightGBM+SHAP 定量揭示库水位急降（贡献 45\%）与强降雨（贡献 30\%）为主控因子，验证"库水位—降雨耦合驱动"机制
    \item \textbf{位移预测}：LSTM 模型测试集 $R^2=0.7916$，RMSE=5.55\,mm，相对误差 0.56\%，达工程应用精度；50次运行集成成功量化认知不确定性（变异系数 0.55\%）
    \item \textbf{概率预警}：基于越限概率的五级预警体系准确率 95.2\%，召回率 88.9\%，F1 分数 87.3\%，平均预警提前量 2.3 天，显著优于传统速率法
  \end{enumerate}

  \end{alertblock}
