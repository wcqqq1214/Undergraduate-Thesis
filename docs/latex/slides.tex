\section{可解释机器学习特征分析}
% ============================================================

\begin{frame}{研究方法：LightGBM + SHAP}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item 直方图算法：连续特征离散化，降低计算量
        \item 叶子优先生长策略（Leaf-wise）
        \item 基于梯度的单边抽样（GOSS）
        \item 互斥特征捆绑（EFB）
      \end{itemize}
      \vspace{0.5em}
      迭代更新公式：
      \begin{equation*}
        F_t(x) = F_{t-1}(x) + \eta f_t(x)
      \end{equation*}
    \end{column}
    \begin{column}{0.50\textwidth}
      {\small
      \textbf{SHAP 可解释性}

      特征 $j$ 的 Shapley 值：
      \begin{equation*}
        \textstyle
        \varphi_j = \sum_{S \subseteq N\setminus\{j\}}
          \frac{|S|!(N\!-\!|S|\!-\!1)!}{N!}
          \bigl[f(S\!\cup\!\{j\})-f(S)\bigr]
      \end{equation*}

      \vspace{0.1em}
      交互值（$j\neq k$）：
      \begin{equation*}
        \varphi_{j,k} = \!\sum_{S \subseteq N\setminus\{j,k\}}
          \frac{|S|!(N\!-\!|S|\!-\!2)!}{2(N\!-\!1)!}
          \nabla_{jk}(S)
      \end{equation*}
      }
    \end{column}
  \end{columns}
\end{frame}

\section{基于 LSTM 的位移预测模型}
% ============================================================

\begin{frame}{LSTM 网络原理}
  LSTM 通过门控机制捕捉长期依赖关系，核心计算如下：

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \textbf{遗忘门}
      \begin{equation*}
        \mathbf{f}_t = \sigma(\mathbf{W}_f [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_f)
      \end{equation*}
      \textbf{输入门}
      \begin{equation*}
        \mathbf{i}_t = \sigma(\mathbf{W}_i [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_i)
      \end{equation*}
      \begin{equation*}
        \tilde{\mathbf{C}}_t = \tanh(\mathbf{W}_C [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_C)
      \end{equation*}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \textbf{记忆单元更新}
      \begin{equation*}
        \mathbf{C}_t = \mathbf{f}_t \odot \mathbf{C}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{C}}_t
      \end{equation*}
      \textbf{输出门}
      \begin{equation*}
        \mathbf{o}_t = \sigma(\mathbf{W}_o [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_o)
      \end{equation*}
      \begin{equation*}
        \mathbf{h}_t = \mathbf{o}_t \odot \tanh(\mathbf{C}_t)
      \end{equation*}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{block}{输入设计}
    MJ9、MJ1、MJ3、ATU4 四点历史累积位移序列（时间窗口 2 天，特征维度 4）$\;\to\;$ 预测 MJ1 趋势项位移
  \end{block}
\end{frame}

\begin{frame}{基于多次运行的概率预测框架}
  \textbf{方法}：对 LSTM 进行 \alert{50 次独立训练}（不同随机种子），收集预测分布

  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{统计量计算}

      均值预测：
      \begin{equation*}
        \bar{y}_t = \frac{1}{50}\sum_{i=1}^{50}\hat{y}_t^{(i)}
      \end{equation*}

      分位数预测区间：
      \begin{equation*}
        \hat{y}_t^{(p)} = \text{Percentile}\!\left(\{\hat{y}_t^{(i)}\},\, p\right)
      \end{equation*}
      构建 50\% 区间（25\%–75\%）和 90\% 区间（5\%–95\%）
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{区间评价指标}

      覆盖概率（PICP）：
      \begin{equation*}
        \text{PICP} = \frac{1}{n}\sum_{i=1}^{n} \mathbb{I}(L_i \leq y_i \leq U_i)
      \end{equation*}

      归一化区间宽度（NMPIW）：
      \begin{equation*}
        \text{NMPIW} = \frac{1}{n \cdot R}\sum_{i=1}^{n}(U_i - L_i)
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{网络结构与超参数}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{模型结构}
      \begin{itemize}
        \item LSTM 层 1：25 个隐藏单元，return\_sequences=True
        \item Dropout：0.3
        \item LSTM 层 2：15 个隐藏单元，L2 正则化（0.002）
        \item Dropout：0.3
        \item 全连接层：15 → 1（输出位移预测值）
        \item 总参数量：5,716
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{训练设置}
      \begin{itemize}
        \item 优化器：Adam，学习率 0.0005
        \item 损失函数：MSE
        \item 批大小：64，训练轮数：40
        \item 数据划分：训练集 80\% / 测试集 20\%
        \item 归一化：MinMaxScaler $[0,1]$
      \end{itemize}
      \vspace{0.5em}
      \begin{block}{硬件}
        NVIDIA GeForce RTX 3060 Laptop GPU\\
        单次训练约 1.5 分钟
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{基于概率预测的滑坡预警方法}
% ============================================================

\begin{frame}{传统预警方法的局限性}
  \begin{enumerate}
    \item \textbf{忽略预测不确定性}：基于点预测判断，预测值接近阈值时易误报或漏报
    \item \textbf{响应时效性不足}：速率/切线角指标属"滞后性指标"，预警窗口极短
    \item \textbf{单一阈值缺乏适应性}：固定阈值难以适应不同变形阶段的动态变化
  \end{enumerate}
  \vspace{0.8em}
  \begin{block}{本章解决方案}
    基于各监测点自身位移统计确定基准速率 $V_0$，将 LSTM \alert{50次独立预测}的月速率分布映射到四级区间，直接统计概率并取最高者为当日预警等级
  \end{block}
\end{frame}

\begin{frame}[shrink=10]{基准速率 $V_0$ 的确定方法}
  \textbf{月速率定义}：累计位移的 30 天滚动增量（mm/月）

  \vspace{0.5em}
  \textbf{三步确定 $V_0$（仅用训练集数据）}：
  \begin{enumerate}
    \item 对训练期累计位移做 30 天滚动差分，得月速率序列 $\{V_i\}$
    \item 剔除 $>90\%$ 分位的加速事件（排除极端工况影响）
    \item 对剩余"匀速变形阶段"样本计算均值 $\bar V$ 与标准差 $\sigma$：
  \end{enumerate}
  \begin{equation*}
    V_0 = 1.5\bar V + 2\sigma
  \end{equation*}

  \begin{block}{四级预警划分}
    \begin{tabular}{clll}
      \toprule
      等级 & 颜色 & 速率范围 & 响应 \\
      \midrule
      I  & 绿色 & $V < V_0$              & 常规监测 \\
      II & 黄色 & $V_0 \leq V < 5V_0$    & 加密监测，现场巡查 \\
      III& 橙色 & $5V_0 \leq V < 10V_0$  & 启动应急预案 \\
      IV & 红色 & $V \geq 10V_0$         & 人员撤离 \\
      \bottomrule
    \end{tabular}
  \end{block}
\end{frame}

\begin{frame}{四级概率判据}
  设第 $i$ 次 LSTM 预测的未来 30 天月速率为 $V_t^{(i)}$，则：
  \begin{equation*}
    \begin{aligned}
      P_{\text{green}}(t)  &= \tfrac{1}{50}\textstyle\sum_{i=1}^{50}\mathbb{I}(V_t^{(i)} < V_0) \\
      P_{\text{yellow}}(t) &= \tfrac{1}{50}\textstyle\sum_{i=1}^{50}\mathbb{I}(V_0 \leq V_t^{(i)} < 5V_0) \\
      P_{\text{orange}}(t) &= \tfrac{1}{50}\textstyle\sum_{i=1}^{50}\mathbb{I}(5V_0 \leq V_t^{(i)} < 10V_0) \\
      P_{\text{red}}(t)    &= \tfrac{1}{50}\textstyle\sum_{i=1}^{50}\mathbb{I}(V_t^{(i)} \geq 10V_0)
    \end{aligned}
  \end{equation*}
  \begin{equation*}
    \text{Level}(t) = \arg\max_{L \in \{\text{green,yellow,orange,red}\}} P_L(t)
  \end{equation*}
  \begin{alertblock}{优势}
    无需分布假设；天然耦合 LSTM 输出；50次集成抗抖动；物理意义明确
  \end{alertblock}
\end{frame}

\begin{frame}{SHAP 归因解释——典型预警事件}
  \textbf{主要驱动因素（2017年9月28日 MJ1 黄色预警，月速率中位数 36.2\,mm/月）}
  \begin{itemize}
    \item \alert{历史位移累积效应}（过去5天位移持续累积）：贡献约 \textbf{40.7\%}
    \item \alert{库水位调度}（30天内 152→167\,m，快速抬升 +14.7\,m）：贡献约 \textbf{36.8\%}
    \item 地下水位变化（30天内上升 +7.46\,m）：贡献约 \textbf{18.4\%}
    \item 气温变化与降雨直接贡献：约 4.1\%
  \end{itemize}
  \vspace{0.3em}
  \begin{block}{结论}
    前期位移累积、库水位快速升高、地下水位补给三大因素合计贡献约 95.9\%，与"主汛期末蓄水启动→位移加速"规律高度吻合
  \end{block}

\section{结论}
% ============================================================

\begin{frame}{主要结论}
  \small
  \begin{enumerate}
    \item \textbf{特征分析}：LightGBM+SHAP 定量揭示库水位消落（39.5\%）与历史位移累积（34.2\%）为主控因子，气温（21.1\%）与地下水位（5.1\%）次之，验证滑坡变形累积驱动机制
    \item \textbf{位移预测}：LSTM 模型 MJ1 测试集 $R^2=0.7916$，RMSE=5.55\,mm，相对误差 0.56\%，达工程应用精度；50次运行变异系数 $<1\%$，成功量化认知不确定性
    \item \textbf{概率预警}：基于 $V_0 = 1.5\bar V + 2\sigma$ 的四级动态预警体系，测试集三点均 100\% 特异性、零误报；训练期 2017 年阶跃段演示中 MJ1 与 MJ3 分别触发 70 天与 129 天黄色预警
  \end{enumerate}
  \vspace{0.2em}
  \begin{alertblock}{核心贡献}
    \small 实现从\textbf{确定性点预测}向\textbf{概率区间预报}的转变，从\textbf{滞后性预警}向\textbf{前瞻性预警}的跨越
  \end{alertblock}
\end{frame}
